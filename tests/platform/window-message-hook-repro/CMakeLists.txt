cmake_minimum_required(VERSION 3.25)
project(MuMainWindowMessageHookReproducer LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This reproducer requires native Windows message dispatch.")
endif()
find_package(SDL3 CONFIG REQUIRED)
add_executable(window-message-hook-repro main.cpp)
target_compile_features(window-message-hook-repro PRIVATE cxx_std_17)
target_link_libraries(window-message-hook-repro PRIVATE SDL3::SDL3-static)
